## Features

- **Four animated states**: Normal, Loading, Success, and Error
- **Smooth blur transitions**: Text and icons blur in/out during state changes
- **Spring animations**: Natural, bouncy transitions using Motion for React
- **Customizable text**: Override default text for each state
- **Icon support**: Show custom icons in normal state

## Usage

```tsx
import { StatusButton } from "@/registry/status-button";
import { TbSend } from "react-icons/tb";

export function Example() {
  const [status, setStatus] = useState<"loading" | "success" | "error">();

  const handleClick = async () => {
    setStatus("loading");
    try {
      await someApiCall();
      setStatus("success");
    } catch {
      setStatus("error");
    }
  };

  return (
    <StatusButton
      variant={status}
      icon={<TbSend />}
      text={{
        loading: "Sending...",
        success: "Sent!",
        error: "Failed",
      }}
      onClick={handleClick}
    >
      Send Message
    </StatusButton>
  );
}
```

## Props

| Prop         | Type                                                     | Default           | Description                     |
| ------------ | -------------------------------------------------------- | ----------------- | ------------------------------- |
| `variant`    | `"loading" \| "success" \| "error"`                      | `undefined`       | The current state               |
| `children`   | `ReactNode`                                              | Required          | Button content for normal state |
| `icon`       | `ReactNode`                                              | `undefined`       | Icon for normal state           |
| `text`       | `{ loading?: string; success?: string; error?: string }` | See below         | Custom text                     |
| `transition` | `Transition`                                             | Spring transition | Motion transition settings      |

### Default Text

- `loading`: "Loading..."
- `success`: "Success"
- `error`: "Error"
