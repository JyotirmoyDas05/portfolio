## Features

- **Scroll progress tracking** with circular indicator
- **Expandable menu** with blur backdrop
- **Filter selection** with smooth animations
- **Keyboard accessible** (Escape to close)
- **Container ref support** for tracking any scrollable element

## Usage

```tsx
import {
  DynamicScrollIslandTOC,
  type TOCItem,
} from "@/components/jyotirmoydas/dynamic-scroll-island-toc";
```

```tsx
const items: TOCItem[] = [
  { name: "All" },
  { name: "Category A", value: "a" },
  { name: "Category B", value: "b" },
];

const scrollRef = useRef<HTMLDivElement>(null);
const [active, setActive] = useState(items[0]);

<DynamicScrollIslandTOC
  data={items}
  value={active}
  setValue={setActive}
  containerRef={scrollRef}
/>;
```

## Props

| Prop           | Type                             | Default                                          | Description                                        |
| -------------- | -------------------------------- | ------------------------------------------------ | -------------------------------------------------- |
| `data`         | `TOCItem[]`                      | required                                         | Array of items to display in the TOC.              |
| `value`        | `TOCItem`                        | `data[0]`                                        | Currently selected item.                           |
| `setValue`     | `(v: TOCItem) => void`           | `undefined`                                      | Callback when selection changes.                   |
| `containerRef` | `RefObject<HTMLElement \| null>` | `undefined`                                      | Ref to scrollable container for progress tracking. |
| `transition`   | `Transition`                     | `{ type: "spring", duration: 0.5, bounce: 0.1 }` | Custom transition settings.                        |
| `layoutPrefix` | `string`                         | `""`                                             | Layout ID prefix for nested instances.             |
| `className`    | `string`                         | `undefined`                                      | Additional CSS classes.                            |

## TOCItem Interface

```tsx
interface TOCItem {
  name: string; // Display name
  value?: string; // Optional filter value (undefined = "All")
}
```
